algoritmo "JogoDaVelhaNaGrandona"
var
   grande: vetor[1..3, 1..3] de caractere
   sub: vetor[1..9, 1..9] de caractere
   jogadasSub: vetor[1..9] de inteiro
   escolhedor, jogadorAtual, vencedorGrande, vencedorSub, celula: caractere
   lg, cg, lp, cp, idx, pos, base: inteiro
   totalJogadas, limiteJogadas, fase: inteiro
   linG, colG, linP, colP, i, j, linhasBloqueadas: inteiro
   jogoEncerrado, jogadaValida, temX, temO: logico
inicio
   para i de 1 ate 3 faca
      para j de 1 ate 3 faca
         grande[i, j] <- " "
      fimpara
   fimpara

   para i de 1 ate 9 faca
      jogadasSub[i] <- 0
      para j de 1 ate 9 faca
         sub[i, j] <- " "
      fimpara
   fimpara

   escolhedor <- "X"
   vencedorGrande <- " "
   totalJogadas <- 0
   limiteJogadas <- 99
   fase <- 0
   jogoEncerrado <- falso

   enquanto nao jogoEncerrado faca
      limpatela
      escreval("=== JOGO DA VELHA NA GRANDONA ===")
      escreval("Jogadas totais: ", totalJogadas, " / ", limiteJogadas)
      escreval("")
      escreval("TABULEIRO GRANDE (A Grandona):")
      escreval("    1   2   3")
      para linG de 1 ate 3 faca
         escreva(linG, " ")
         para colG de 1 ate 3 faca
            se grande[linG, colG] = " " entao
               celula <- "."
            senao
               celula <- grande[linG, colG]
            fimse
            escreva(" ", celula, " ")
            se colG < 3 entao
               escreva("|")
            fimse
         fimpara
         escreval("")
         se linG < 3 entao
            escreval("   ---+---+---")
         fimse
      fimpara

      escreval("")
      escreval("VISAO COMPLETA DOS SUBTABULEIROS:")
      para linG de 1 ate 3 faca
         para linP de 1 ate 3 faca
            para colG de 1 ate 3 faca
               idx <- (linG - 1) * 3 + colG
               para colP de 1 ate 3 faca
                  pos <- (linP - 1) * 3 + colP
                  se sub[idx, pos] = " " entao
                     celula <- "."
                  senao
                     celula <- sub[idx, pos]
                  fimse
                  escreva(" ", celula, " ")
               fimpara
               se colG < 3 entao
                  escreva(" | ")
               fimse
            fimpara
            escreval("")
         fimpara
         se linG < 3 entao
            escreval("---------+-----------+---------")
         fimse
      fimpara
      escreval("")

      se fase = 0 entao
         escreval("Vez de [", escolhedor, "] escolher uma casa disponivel na GRANDONA.")
         jogadaValida <- falso
         enquanto nao jogadaValida faca
            escreva("Linha da Grandona (1 a 3): ")
            leia(lg)
            escreva("Coluna da Grandona (1 a 3): ")
            leia(cg)

            se (lg >= 1) e (lg <= 3) e (cg >= 1) e (cg <= 3) entao
               se grande[lg, cg] = " " entao
                  jogadaValida <- verdadeiro
               senao
                  escreval("Essa casa grande ja foi conquistada por ", grande[lg, cg], "! Escolha outra.")
               fimse
            senao
               escreval("Coordenadas invalidas! Digite valores entre 1 e 3.")
            fimse
         fimenquanto

         idx <- (lg - 1) * 3 + cg
         se jogadasSub[idx] = 9 entao
            jogadasSub[idx] <- 0
            para j de 1 ate 9 faca
               sub[idx, j] <- " "
            fimpara
         fimse

         jogadorAtual <- escolhedor
         fase <- 1
      senao
         idx <- (lg - 1) * 3 + cg
         escreval("Disputando subtabuleiro (", lg, ", ", cg, ") | Vez de: [", jogadorAtual, "]")
         jogadaValida <- falso
         enquanto nao jogadaValida faca
            escreva("Linha do subtabuleiro (1 a 3): ")
            leia(lp)
            escreva("Coluna do subtabuleiro (1 a 3): ")
            leia(cp)

            se (lp >= 1) e (lp <= 3) e (cp >= 1) e (cp <= 3) entao
               pos <- (lp - 1) * 3 + cp
               se sub[idx, pos] = " " entao
                  jogadaValida <- verdadeiro
               senao
                  escreval("Posicao ja ocupada! Escolha outra.")
               fimse
            senao
               escreval("Coordenadas invalidas! Digite valores entre 1 e 3.")
            fimse
         fimenquanto

         sub[idx, pos] <- jogadorAtual
         totalJogadas <- totalJogadas + 1
         jogadasSub[idx] <- jogadasSub[idx] + 1

         vencedorSub <- " "
         para i de 1 ate 3 faca
            base <- (i - 1) * 3
            se (sub[idx, base + 1] <> " ") e (sub[idx, base + 1] = sub[idx, base + 2]) e (sub[idx, base + 2] = sub[idx, base + 3]) entao
               vencedorSub <- sub[idx, base + 1]
            fimse
            se (sub[idx, i] <> " ") e (sub[idx, i] = sub[idx, i + 3]) e (sub[idx, i + 3] = sub[idx, i + 6]) entao
               vencedorSub <- sub[idx, i]
            fimse
         fimpara

         se (sub[idx, 1] <> " ") e (sub[idx, 1] = sub[idx, 5]) e (sub[idx, 5] = sub[idx, 9]) entao
            vencedorSub <- sub[idx, 1]
         fimse
         se (sub[idx, 3] <> " ") e (sub[idx, 3] = sub[idx, 5]) e (sub[idx, 5] = sub[idx, 7]) entao
            vencedorSub <- sub[idx, 3]
         fimse

         se (vencedorSub <> " ") ou (jogadasSub[idx] = 9) entao
            se vencedorSub <> " " entao
               grande[lg, cg] <- vencedorSub
            fimse
            fase <- 0

            vencedorGrande <- " "
            para i de 1 ate 3 faca
               se (grande[i, 1] <> " ") e (grande[i, 1] = grande[i, 2]) e (grande[i, 2] = grande[i, 3]) entao
                  vencedorGrande <- grande[i, 1]
               fimse
               se (grande[1, i] <> " ") e (grande[1, i] = grande[2, i]) e (grande[2, i] = grande[3, i]) entao
                  vencedorGrande <- grande[1, i]
               fimse
            fimpara

            se (grande[1, 1] <> " ") e (grande[1, 1] = grande[2, 2]) e (grande[2, 2] = grande[3, 3]) entao
               vencedorGrande <- grande[1, 1]
            fimse
            se (grande[1, 3] <> " ") e (grande[1, 3] = grande[2, 2]) e (grande[2, 2] = grande[3, 1]) entao
               vencedorGrande <- grande[1, 3]
            fimse

            linhasBloqueadas <- 0
            para i de 1 ate 3 faca
               temX <- falso
               temO <- falso
               para j de 1 ate 3 faca
                  se grande[i, j] = "X" entao
                     temX <- verdadeiro
                  fimse
                  se grande[i, j] = "O" entao
                     temO <- verdadeiro
                  fimse
               fimpara
               se temX e temO entao
                  linhasBloqueadas <- linhasBloqueadas + 1
               fimse
            fimpara

            para j de 1 ate 3 faca
               temX <- falso
               temO <- falso
               para i de 1 ate 3 faca
                  se grande[i, j] = "X" entao
                     temX <- verdadeiro
                  fimse
                  se grande[i, j] = "O" entao
                     temO <- verdadeiro
                  fimse
               fimpara
               se temX e temO entao
                  linhasBloqueadas <- linhasBloqueadas + 1
               fimse
            fimpara

            temX <- falso
            temO <- falso
            para i de 1 ate 3 faca
               se grande[i, i] = "X" entao
                  temX <- verdadeiro
               fimse
               se grande[i, i] = "O" entao
                  temO <- verdadeiro
               fimse
            fimpara
            se temX e temO entao
               linhasBloqueadas <- linhasBloqueadas + 1
            fimse

            temX <- falso
            temO <- falso
            para i de 1 ate 3 faca
               se grande[i, 4 - i] = "X" entao
                  temX <- verdadeiro
               fimse
               se grande[i, 4 - i] = "O" entao
                  temO <- verdadeiro
               fimse
            fimpara
            se temX e temO entao
               linhasBloqueadas <- linhasBloqueadas + 1
            fimse

            se (vencedorGrande <> " ") ou (linhasBloqueadas = 8) ou (totalJogadas >= limiteJogadas) entao
               jogoEncerrado <- verdadeiro
            senao
               se escolhedor = "X" entao
                  escolhedor <- "O"
               senao
                  escolhedor <- "X"
               fimse
            fimse
         senao
            se jogadorAtual = "X" entao
               jogadorAtual <- "O"
            senao
               jogadorAtual <- "X"
            fimse
            se totalJogadas >= limiteJogadas entao
               jogoEncerrado <- verdadeiro
            fimse
         fimse
      fimse
   fimenquanto

   limpatela
   escreval("=== JOGO DA VELHA NA GRANDONA ===")
   escreval("Jogadas totais: ", totalJogadas, " / ", limiteJogadas)
   escreval("")
   escreval("TABULEIRO GRANDE (A Grandona):")
   escreval("    1   2   3")
   para linG de 1 ate 3 faca
      escreva(linG, " ")
      para colG de 1 ate 3 faca
         se grande[linG, colG] = " " entao
            celula <- "."
         senao
            celula <- grande[linG, colG]
         fimse
         escreva(" ", celula, " ")
         se colG < 3 entao
            escreva("|")
         fimse
      fimpara
      escreval("")
      se linG < 3 entao
         escreval("   ---+---+---")
      fimse
   fimpara
   escreval("")

   se vencedorGrande <> " " entao
      escreval("FIM DE JOGO! O jogador [", vencedorGrande, "] venceu a Grandona!")
   senao
      se totalJogadas >= limiteJogadas entao
         escreval("VELHA! Limite de ", limiteJogadas, " jogadas atingido. Ninguem venceu.")
      senao
         escreval("VELHA! Nao ha mais vitoria possivel no tabuleiro grande. Ninguem venceu.")
      fimse
   fimse
fimalgoritmo
